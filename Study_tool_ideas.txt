Minimal SQLite Schema:
-- Core vocabulary entry
CREATE TABLE vocab (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    language TEXT CHECK(language IN ('ja', 'ko')),
    lemma TEXT NOT NULL,
    surface TEXT NOT NULL,
    reading TEXT NOT NULL,
    traditional_hanzi TEXT,
    pinyin TEXT,
    category TEXT,            -- onyomi, kunyomi, jukujigo, etc.
    meaning_gloss TEXT,
    UNIQUE(language, lemma)
);

-- SRS review tracking (Anki SM-2 / Leitner state)
CREATE TABLE srs_card (
    vocab_id INTEGER PRIMARY KEY,
    reading_state INTEGER DEFAULT 0,  -- 0: unseen, 1: learning, 2: mastered
    meaning_state INTEGER DEFAULT 1,  -- 1: recognized via Hanzi, 0: unknown
    box INTEGER DEFAULT 1,            -- Current review bucket
    ease_factor REAL DEFAULT 2.5,
    interval_days INTEGER DEFAULT 0,
    due_timestamp INTEGER,
    FOREIGN KEY(vocab_id) REFERENCES vocab(id) ON DELETE CASCADE
);

-- Links words back to specific song timestamps
CREATE TABLE song_occurrences (
    id INTEGER PRIMARY KEY AUTOINCREMENT,
    vocab_id INTEGER,
    group_name TEXT,
    song_title TEXT,
    singer_name TEXT,
    lyric_line TEXT,
    start_chunk INTEGER,
    end_chunk INTEGER,
    FOREIGN KEY(vocab_id) REFERENCES vocab(id) ON DELETE CASCADE
);
